# A process name containing a comma is quoted (RFC 4180)
$ procmon -p 13821 -n 1 -c
timestamp,pid,name,cpu_percent,memory_mb,uptime_s
2026-11-17 19:01:24,13821,"a,b",,2.0,0
[exit code: 0]

# Parsed back with Python's csv module (6 fields per row):

# -p and -N combined; a PID found by both is monitored only once
$ procmon -p 1,13827 -N pm_target -n 1

2026-11-17 19:01:26        1  systemd             N/A         16.5  03:31:42

[exit code: 0]

# Unknown name mixed with a valid one: warning, valid one still monitored
$ procmon -N no_such_proc,pm_target -n 1
