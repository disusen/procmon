$ procmon -p 1 -i 1   (then SIGINT)

# A process name containing a comma is quoted (RFC 4180)
$ procmon -p 2303 -n 1 -c
timestamp,pid,name,cpu_percent,memory_mb,uptime_s
2026-11-17 22:14:56,2303,"a,b",,2.0,0
[exit code: 0]

# Parsed back with Python's csv module (6 fields per row):
['timestamp', 'pid', 'name', 'cpu_percent', 'memory_mb', 'uptime_s']
['2026-11-17 22:14:56', '1', 'systemd', '', '15.9', '431']
['2026-11-17 22:14:56', '2303', 'a,b', '', '2.0', '0']

$ procmon -N pm_target -i 1 -n 2
Timestamp                PID  Name              CPU %  Memory (MB)  Uptime
2026-11-17 22:14:56     2309  pm_target           N/A          2.0  00:00:00
2026-11-17 22:14:56     2310  pm_target           N/A          2.0  00:00:00
2026-11-17 22:14:57     2309  pm_target           0.0          2.0  00:00:01
2026-11-17 22:14:57     2310  pm_target           0.0          2.0  00:00:01
[exit code: 0]

# -p and -N combined; a PID found by both is monitored only once
$ procmon -p 1,2309 -N pm_target -n 1
Timestamp                PID  Name              CPU %  Memory (MB)  Uptime
2026-11-17 22:14:57        1  systemd             N/A         15.9  00:07:12
2026-11-17 22:14:57     2309  pm_target           N/A          2.0  00:00:01
2026-11-17 22:14:57     2310  pm_target           N/A          2.0  00:00:01
[exit code: 0]

# Unknown name mixed with a valid one: warning, valid one still monitored
$ procmon -N no_such_proc,pm_target -n 1
warning: no process named 'no_such_proc'

2026-11-17 22:14:57     2310  pm_target           N/A          2.0  00:00:01
[exit code: 0]

# Only unknown names: nothing to monitor

[exit code: 1]
